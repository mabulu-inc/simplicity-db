<CardGrid>
  <Card title="Thin over pg" icon="pencil">

  </Card>
  <Card title="Schema-agnostic" icon="approve-check">

  </Card>
  <Card title="Typed errors" icon="setting">
    Map Postgres SQLSTATE codes to user-facing messages and HTTP statuses without restating the copy in every service.
  </Card>
